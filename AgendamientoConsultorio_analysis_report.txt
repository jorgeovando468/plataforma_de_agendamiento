// auth.js (corregido)
const express = require('express');
const bcrypt = require('bcrypt');
const jwt = require('jsonwebtoken');
const db = require('./db');
require('dotenv').config();

const router = express.Router();

/**
 * POST /api/auth/login
 * Autentica al administrador.
 * Body: { usuario: string, contraseña: string }
 * Respuesta: { status, message, data: { token } } o error.
 */
router.post('/login', async (req, res) => {
    const { usuario, contraseña } = req.body;

    if (!usuario || !contraseña) {
        return res.status(400).json({ status: 'error', message: 'Usuario y contraseña son requeridos.' });
    }

    try {
        // Ejecutar consulta y extraer rows de manera segura
        const result = await db.execute('SELECT * FROM admin WHERE usuario = ?', [usuario]);
        const rows = result[0] || [];  // result[0] es rows, o array vacío si falla

        console.log('Resultado de consulta:', rows);  // Para depurar

        if (rows.length === 0) {
            return res.status(401).json({ status: 'error', message: 'Credenciales inválidas.' });
        }

        const admin = rows[0];

        // Verificar contraseña con bcrypt
        const isValidPassword = await bcrypt.compare(contraseña, admin.contraseña);
        if (!isValidPassword) {
            return res.status(401).json({ status: 'error', message: 'Credenciales inválidas.' });
        }

        // Generar JWT
        const token = jwt.sign({ id: admin.id, usuario: admin.usuario }, process.env.JWT_SECRET, { expiresIn: '1h' });

        res.json({ status: 'success', message: 'Login exitoso.', data: { token } });
    } catch (error) {
        console.error('Error en login:', error);
        res.status(500).json({ status: 'error', message: 'Error interno del servidor.' });
    }
});

module.exports = router;
